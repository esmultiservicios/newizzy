-- CONFIGURACION CENTRAL - DB_MAIN
-- ============================================================
--
-- EJECUTAR UNICAMENTE EN DB_MAIN
--
-- Este script:
-- 1. Crea historial de configuracion.
-- 2. Crea accesos para Pantalla Cocina.
-- 3. Crea pruebas de conexion.
-- 4. Agrega oculto_vista a dispositivos SI la tabla existe.
--
-- NO BORRA INFORMACION.
-- PUEDE EJECUTARSE MAS DE UNA VEZ.
-- ============================================================

SET @db_actual := DATABASE();


-- ============================================================
-- 1. HISTORIAL DE CONFIGURACION RESTAURANTE
-- ============================================================

CREATE TABLE IF NOT EXISTS restaurante_configuracion_historial (
    historial_id BIGINT NOT NULL AUTO_INCREMENT,
    server_customers_id INT NOT NULL DEFAULT 0,
    empresa_id INT NOT NULL,
    users_id INT NOT NULL DEFAULT 0,
    colaborador_id INT NOT NULL DEFAULT 0,
    categoria VARCHAR(40) NOT NULL DEFAULT 'general',
    resumen VARCHAR(255) NOT NULL DEFAULT '',
    cambios_json LONGTEXT NULL,
    fecha_registro DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,

    PRIMARY KEY (historial_id),

    KEY idx_rest_cfg_hist_empresa (
        server_customers_id,
        empresa_id,
        fecha_registro
    )
)
ENGINE=InnoDB
DEFAULT CHARSET=utf8mb4
COLLATE=utf8mb4_general_ci;


-- ============================================================
-- 2. ACCESOS PARA PANTALLA COCINA INDEPENDIENTE
-- ============================================================

CREATE TABLE IF NOT EXISTS restaurante_pantalla_accesos (
    acceso_id BIGINT NOT NULL AUTO_INCREMENT,

    server_customers_id INT NOT NULL,
    empresa_id INT NOT NULL,

    tipo VARCHAR(20) NOT NULL DEFAULT 'cocina',

    token_hash CHAR(64) NOT NULL,
    token_cifrado TEXT NOT NULL,

    activo TINYINT(1) NOT NULL DEFAULT 0,

    fecha_registro DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion DATETIME NULL DEFAULT NULL,
    fecha_regeneracion DATETIME NULL DEFAULT NULL,

    PRIMARY KEY (acceso_id),

    UNIQUE KEY uq_rest_pantalla_empresa_tipo (
        server_customers_id,
        empresa_id,
        tipo
    ),

    UNIQUE KEY uq_rest_pantalla_token_hash (
        token_hash
    ),

    KEY idx_rest_pantalla_resolver (
        token_hash,
        tipo,
        activo
    ),

    KEY idx_rest_pantalla_cliente (
        server_customers_id,
        empresa_id
    )
)
ENGINE=InnoDB
DEFAULT CHARSET=utf8mb4
COLLATE=utf8mb4_general_ci;


-- ============================================================
-- 3. PRUEBAS DE CONEXION DE PANTALLA COCINA
-- ============================================================

CREATE TABLE IF NOT EXISTS restaurante_pantalla_pruebas (
    prueba_id BIGINT NOT NULL AUTO_INCREMENT,
    acceso_id BIGINT NOT NULL,
    mensaje VARCHAR(160) NOT NULL DEFAULT 'Prueba de conexión IZZY',
    fecha_registro DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_expira DATETIME NOT NULL,

    PRIMARY KEY (prueba_id),

    KEY idx_rest_pantalla_prueba (
        acceso_id,
        fecha_expira
    )
)
ENGINE=InnoDB
DEFAULT CHARSET=utf8mb4
COLLATE=utf8mb4_general_ci;


-- ============================================================
-- 4. OCULTAR DISPOSITIVOS DE COCINA DE FORMA PERSISTENTE
-- ============================================================
--
-- IMPORTANTE:
-- restaurante_pantalla_dispositivos debe pertenecer a DB_MAIN.
--
-- Si la tabla existe:
--     agrega oculto_vista si hace falta.
--
-- Si la tabla NO existe:
--     NO genera error #1146.
-- ============================================================

SET @existe_tabla_dispositivos := (
    SELECT COUNT(*)
    FROM INFORMATION_SCHEMA.TABLES
    WHERE TABLE_SCHEMA = @db_actual
      AND TABLE_NAME = 'restaurante_pantalla_dispositivos'
);


SET @existe_oculto_vista := (
    SELECT COUNT(*)
    FROM INFORMATION_SCHEMA.COLUMNS
    WHERE TABLE_SCHEMA = @db_actual
      AND TABLE_NAME = 'restaurante_pantalla_dispositivos'
      AND COLUMN_NAME = 'oculto_vista'
);


SET @sql_dispositivos := IF(
    @existe_tabla_dispositivos = 0,

    'SELECT
        ''ATENCION: restaurante_pantalla_dispositivos NO existe en esta base. No se realizo ningun cambio.'' AS resultado',

    IF(
        @existe_oculto_vista = 0,

        'ALTER TABLE restaurante_pantalla_dispositivos
         ADD COLUMN oculto_vista TINYINT(1) NOT NULL DEFAULT 0
         AFTER activo',

        'SELECT
            ''restaurante_pantalla_dispositivos.oculto_vista ya existe'' AS resultado'
    )
);


PREPARE stmt_dispositivos FROM @sql_dispositivos;

EXECUTE stmt_dispositivos;

DEALLOCATE PREPARE stmt_dispositivos;


-- ============================================================
-- 5. VERIFICAR DISPOSITIVOS
-- ============================================================
--
-- Solo ejecuta SELECT si la tabla realmente existe.
-- ============================================================

SET @sql_ver_dispositivos := IF(
    @existe_tabla_dispositivos > 0,

    'SELECT
        dispositivo_id,
        nombre,
        activo,
        oculto_vista,
        fecha_vinculacion,
        fecha_desvinculacion
     FROM restaurante_pantalla_dispositivos
     ORDER BY
        activo DESC,
        oculto_vista ASC,
        dispositivo_id DESC',

    'SELECT
        ''No se verificaron dispositivos porque restaurante_pantalla_dispositivos no existe en esta base.'' AS resultado'
);


PREPARE stmt_ver_dispositivos FROM @sql_ver_dispositivos;

EXECUTE stmt_ver_dispositivos;

DEALLOCATE PREPARE stmt_ver_dispositivos;


-- ============================================================
-- 6. VERIFICACION DE ACCESOS
-- ============================================================

SELECT
    acceso_id,
    server_customers_id,
    empresa_id,
    tipo,
    activo,
    fecha_regeneracion
FROM restaurante_pantalla_accesos
ORDER BY acceso_id DESC;


-- ============================================================
-- 7. VERIFICACION FINAL DB_MAIN
-- ============================================================

SELECT
    DATABASE() AS base_actual,
    @existe_tabla_dispositivos AS existe_tabla_dispositivos,
    (
        SELECT COUNT(*)
        FROM INFORMATION_SCHEMA.TABLES
        WHERE TABLE_SCHEMA = DATABASE()
          AND TABLE_NAME = 'restaurante_configuracion_historial'
    ) AS existe_historial,
    (
        SELECT COUNT(*)
        FROM INFORMATION_SCHEMA.TABLES
        WHERE TABLE_SCHEMA = DATABASE()
          AND TABLE_NAME = 'restaurante_pantalla_accesos'
    ) AS existe_accesos,
    (
        SELECT COUNT(*)
        FROM INFORMATION_SCHEMA.TABLES
        WHERE TABLE_SCHEMA = DATABASE()
          AND TABLE_NAME = 'restaurante_pantalla_pruebas'
    ) AS existe_pruebas;

-- IZZY - Instalación/actualización de facturación recurrente (MySQL 8+)
-- 1=Activa, 2=Cancelada, 3=Finalizada

CREATE TABLE IF NOT EXISTS facturas_recurrentes (
  rec_id INT NOT NULL AUTO_INCREMENT,
  empresa_id INT NOT NULL,
  clientes_id INT NOT NULL,
  colaboradores_id INT NOT NULL,
  tipo_documento TINYINT NOT NULL COMMENT '0=Normal, 1=Proforma',
  tipo_factura TINYINT NOT NULL DEFAULT 2 COMMENT 'Siempre 2=Credito',
  notas VARCHAR(255) NOT NULL DEFAULT '',
  fecha_dolar DATE NOT NULL,
  exoneracion_orden VARCHAR(100) DEFAULT NULL,
  exoneracion_constancia VARCHAR(100) DEFAULT NULL,
  exoneracion_sag VARCHAR(100) DEFAULT NULL,
  exoneracion_orden_interno VARCHAR(100) DEFAULT NULL,
  periodicidad ENUM('once','daily','weekly','monthly') NOT NULL DEFAULT 'monthly',
  dia_mes TINYINT DEFAULT NULL COMMENT 'Día original para recurrencia mensual',
  start_at DATETIME NOT NULL,
  next_run_at DATETIME NOT NULL,
  until_at DATE DEFAULT NULL,
  estado TINYINT NOT NULL DEFAULT 1 COMMENT '1=Activa, 2=Cancelada, 3=Finalizada',
  enviar_correo TINYINT NOT NULL DEFAULT 1 COMMENT '1=Enviar al cliente, 2=No enviar',
  ultimo_facturas_id INT DEFAULT NULL,
  last_run_at DATETIME DEFAULT NULL,
  ultimo_error TEXT DEFAULT NULL,
  usuario_crea INT NOT NULL,
  fecha_crea DATETIME NOT NULL,
  PRIMARY KEY (rec_id),
  KEY idx_recurrente_pendiente (estado, next_run_at),
  KEY idx_recurrente_empresa (empresa_id, rec_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

-- MySQL no soporta ADD COLUMN IF NOT EXISTS en todas sus variantes.
-- Este procedimiento permite ejecutar el instalador aunque las tablas ya existan.
DROP PROCEDURE IF EXISTS izzy_agregar_columna;
DELIMITER $$
CREATE PROCEDURE izzy_agregar_columna(
  IN p_tabla VARCHAR(64),
  IN p_columna VARCHAR(64),
  IN p_definicion TEXT
)
BEGIN
  IF NOT EXISTS (
    SELECT 1
    FROM information_schema.COLUMNS
    WHERE TABLE_SCHEMA = DATABASE()
      AND TABLE_NAME = p_tabla
      AND COLUMN_NAME = p_columna
  ) THEN
    SET @sql_izzy = CONCAT('ALTER TABLE `', p_tabla, '` ADD COLUMN `', p_columna, '` ', p_definicion);
    PREPARE stmt_izzy FROM @sql_izzy;
    EXECUTE stmt_izzy;
    DEALLOCATE PREPARE stmt_izzy;
  END IF;
END$$
DELIMITER ;

CALL izzy_agregar_columna('facturas_recurrentes', 'dia_mes', 'TINYINT DEFAULT NULL AFTER `periodicidad`');
CALL izzy_agregar_columna('facturas_recurrentes', 'enviar_correo', 'TINYINT NOT NULL DEFAULT 1 AFTER `estado`');
CALL izzy_agregar_columna('facturas_recurrentes', 'ultimo_facturas_id', 'INT DEFAULT NULL AFTER `enviar_correo`');
CALL izzy_agregar_columna('facturas_recurrentes', 'last_run_at', 'DATETIME DEFAULT NULL AFTER `ultimo_facturas_id`');
CALL izzy_agregar_columna('facturas_recurrentes', 'ultimo_error', 'TEXT DEFAULT NULL AFTER `last_run_at`');

/* ============================================================
   MIGRACIÓN SEGURA - INVERSIÓN / REPOSICIÓN
   ------------------------------------------------------------
   Objetivo:
   1. Agregar columna es_inversion en categoria_gastos si no existe.
   2. Marcar categorías que tengan INVERSION / INVERSIÓN / REPOSICION / REPOSICIÓN.
   3. Actualizar egresos de inversión solo si existe una categoría marcada como inversión.
   4. Agregar columna es_inversion en cuentas si no existe.
   5. Marcar cuentas como inversión SOLO si el nombre contiene inversión/reposición.
      Si no existe una cuenta de inversión, no marca ninguna.
   ============================================================ */


/* ============================================================
   1. AGREGAR COLUMNA es_inversion EN categoria_gastos SI NO EXISTE
   ============================================================ */

SET @existe_columna_categoria := (
    SELECT COUNT(*)
    FROM INFORMATION_SCHEMA.COLUMNS
    WHERE TABLE_SCHEMA = DATABASE()
      AND TABLE_NAME = 'categoria_gastos'
      AND COLUMN_NAME = 'es_inversion'
);

SET @sql_categoria := IF(
    @existe_columna_categoria = 0,
    'ALTER TABLE categoria_gastos ADD COLUMN es_inversion TINYINT(1) NOT NULL DEFAULT 0 COMMENT ''1 = No se cuenta como gasto; se clasifica como inversión/reposición''',
    'SELECT ''La columna categoria_gastos.es_inversion ya existe'' AS mensaje'
);

PREPARE stmt_categoria FROM @sql_categoria;
EXECUTE stmt_categoria;
DEALLOCATE PREPARE stmt_categoria;


/* ============================================================
   2. MARCAR CATEGORÍAS DE INVERSIÓN / REPOSICIÓN
   ============================================================ */

UPDATE categoria_gastos
SET es_inversion = 1
WHERE UPPER(nombre) LIKE '%INVERSION%'
   OR UPPER(nombre) LIKE '%INVERSIÓN%'
   OR UPPER(nombre) LIKE '%REPOSICION%'
   OR UPPER(nombre) LIKE '%REPOSICIÓN%';


/* ============================================================
   3. ACTUALIZAR EGRESOS DE INVERSIÓN SOLO SI EXISTE CATEGORÍA
      MARCADA COMO INVERSIÓN
   ============================================================ */

UPDATE egresos
SET categoria_gastos_id = (
    SELECT cg.categoria_gastos_id
    FROM categoria_gastos cg
    WHERE cg.es_inversion = 1
    ORDER BY cg.categoria_gastos_id ASC
    LIMIT 1
)
WHERE estado = 1
  AND tipo_egreso = 2
  AND UPPER(observacion) LIKE '%INVERSION%'
  AND EXISTS (
      SELECT 1
      FROM categoria_gastos cg2
      WHERE cg2.es_inversion = 1
  );


/* ============================================================
   4. AGREGAR COLUMNA es_inversion EN cuentas SI NO EXISTE
   ============================================================ */

SET @existe_columna_cuenta := (
    SELECT COUNT(*)
    FROM INFORMATION_SCHEMA.COLUMNS
    WHERE TABLE_SCHEMA = DATABASE()
      AND TABLE_NAME = 'cuentas'
      AND COLUMN_NAME = 'es_inversion'
);

SET @sql_cuenta := IF(
    @existe_columna_cuenta = 0,
    'ALTER TABLE cuentas ADD COLUMN es_inversion TINYINT(1) NOT NULL DEFAULT 0 COMMENT ''1 = Cuenta usada para inversión/reposición''',
    'SELECT ''La columna cuentas.es_inversion ya existe'' AS mensaje'
);

PREPARE stmt_cuenta FROM @sql_cuenta;
EXECUTE stmt_cuenta;
DEALLOCATE PREPARE stmt_cuenta;


/* ============================================================
   5. MARCAR CUENTAS DE INVERSIÓN / REPOSICIÓN SOLO SI EXISTEN
      NO SE USA cuentas_id FIJO.
   ============================================================ */

UPDATE cuentas
SET es_inversion = 1
WHERE UPPER(nombre) LIKE '%INVERSION%'
   OR UPPER(nombre) LIKE '%INVERSIÓN%'
   OR UPPER(nombre) LIKE '%REPOSICION%'
   OR UPPER(nombre) LIKE '%REPOSICIÓN%';


/* ============================================================
   6. VERIFICACIÓN FINAL
   ============================================================ */

SELECT 
    cuentas_id,
    codigo,
    nombre,
    es_inversion
FROM cuentas
WHERE es_inversion = 1;

SELECT 
    categoria_gastos_id,
    nombre,
    es_inversion
FROM categoria_gastos
WHERE es_inversion = 1;

-- Promoción (encabezado)
CREATE TABLE promociones (
  promo_id            INT AUTO_INCREMENT PRIMARY KEY,
  empresa_id          INT NOT NULL,
  nombre              VARCHAR(120) NOT NULL,
  descripcion         VARCHAR(255) NULL,

  -- 'PORC' = % ; 'MONTO' = rebaja fija (moneda)
  tipo_descuento      ENUM('PORC','MONTO') NOT NULL,
  valor               DECIMAL(12,2) NOT NULL,

  -- Vigencia por días (solo fecha)
  fecha_inicio        DATE NOT NULL,
  fecha_fin           DATE NOT NULL,

  -- Ventana diaria opcional (aplica cada día del rango si se define)
  hora_inicio         TIME NULL,
  hora_fin            TIME  NULL,

  -- Días de la semana opcionales
  dias_semana         SET('mon','tue','wed','thu','fri','sat','sun') NULL,

  prioridad           INT NOT NULL DEFAULT 0,
  aplica_a            ENUM('PRODUCTO','CATEGORIA','TODOS') NOT NULL,
  acumula_con_mayoreo TINYINT(1) NOT NULL DEFAULT 0,
  estado              TINYINT(1) NOT NULL DEFAULT 1,
  creado_por          INT NULL,
  creado_en           TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

  INDEX ix_promos_vigencia (empresa_id, estado, fecha_inicio, fecha_fin, prioridad)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Relación con productos concretos
CREATE TABLE promo_productos (
  promo_id     INT NOT NULL,
  producto_id  INT NOT NULL,
  PRIMARY KEY (promo_id, producto_id),
  CONSTRAINT fk_pp_promo FOREIGN KEY (promo_id) REFERENCES promociones(promo_id) ON DELETE CASCADE,
  CONSTRAINT fk_pp_prod  FOREIGN KEY (producto_id) REFERENCES productos(productos_id) ON DELETE CASCADE,
  INDEX ix_pp_prod (producto_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Relación con categorías completas
CREATE TABLE promo_categorias (
  promo_id     INT NOT NULL,
  categoria_id INT NOT NULL,
  PRIMARY KEY (promo_id, categoria_id),
  CONSTRAINT fk_pc_promo FOREIGN KEY (promo_id) REFERENCES promociones(promo_id) ON DELETE CASCADE,
  -- asumiendo que tienes tabla categorias(id)
  INDEX ix_pc_cat (categoria_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;


CREATE OR REPLACE VIEW v_promos_vigentes_producto AS
SELECT
    pr.promo_id, pr.empresa_id, p.productos_id AS producto_id,
    pr.tipo_descuento, pr.valor, pr.prioridad, pr.acumula_con_mayoreo
FROM promociones pr
JOIN promo_productos pp   ON pp.promo_id = pr.promo_id
JOIN productos p          ON p.productos_id = pp.producto_id
WHERE pr.estado=1
  AND NOW() BETWEEN pr.fecha_inicio AND pr.fecha_fin
  AND (pr.hora_inicio IS NULL OR pr.hora_fin IS NULL OR TIME(NOW()) BETWEEN pr.hora_inicio AND pr.hora_fin)
  AND (pr.dias_semana IS NULL OR FIND_IN_SET(LOWER(DAYNAME(NOW())), pr.dias_semana) > 0)

UNION ALL
SELECT
    pr.promo_id, pr.empresa_id, p.productos_id AS producto_id,
    pr.tipo_descuento, pr.valor, pr.prioridad, pr.acumula_con_mayoreo
FROM promociones pr
JOIN promo_categorias pc ON pc.promo_id = pr.promo_id
JOIN productos p         ON p.categoria_id = pc.categoria_id
WHERE pr.estado=1
  AND NOW() BETWEEN pr.fecha_inicio AND pr.fecha_fin
  AND (pr.hora_inicio IS NULL OR pr.hora_fin IS NULL OR TIME(NOW()) BETWEEN pr.hora_inicio AND pr.hora_fin)
  AND (pr.dias_semana IS NULL OR FIND_IN_SET(LOWER(DAYNAME(NOW())), pr.dias_semana) > 0)

UNION ALL
-- Promos globales (TODOS los productos)
SELECT
    pr.promo_id, pr.empresa_id, p.productos_id AS producto_id,
    pr.tipo_descuento, pr.valor, pr.prioridad, pr.acumula_con_mayoreo
FROM promociones pr
JOIN productos p ON p.empresa_id = pr.empresa_id
WHERE pr.estado=1
  AND pr.aplica_a='TODOS'
  AND NOW() BETWEEN pr.fecha_inicio AND pr.fecha_fin
  AND (pr.hora_inicio IS NULL OR pr.hora_fin IS NULL OR TIME(NOW()) BETWEEN pr.hora_inicio AND pr.hora_fin)
  AND (pr.dias_semana IS NULL OR FIND_IN_SET(LOWER(DAYNAME(NOW())), pr.dias_semana) > 0);

# Nuevo query para la vista INVENTARIO ARCHIVO llenarDataTableInventarioTrasnferencia.php
SELECT
	m.almacen_id,
	m.movimientos_id AS 'movimientos_id',
	p.barCode AS 'barCode',
	p.nombre AS 'producto',
	me.nombre AS 'medida',
	SUM(m.cantidad_entrada) AS 'entrada',
	SUM(m.cantidad_salida) AS 'salida',
	(
		SUM(m.cantidad_entrada) - SUM(m.cantidad_salida)
	) AS 'saldo',
	bo.nombre AS 'bodega',
	DATE_FORMAT(
		p.fecha_registro,
		'%d/%m/%Y %H:%i:%s'
	) AS 'fecha_registro',
	p.productos_id AS 'productos_id',
	p.id_producto_superior
FROM
	movimientos AS m
RIGHT JOIN productos AS p ON m.productos_id = p.productos_id
LEFT JOIN medida AS me ON p.medida_id = me.medida_id
LEFT JOIN almacen AS bo ON m.almacen_id = bo.almacen_id
WHERE
	p.estado = 1
GROUP BY
	p.productos_id, m.almacen_id
ORDER BY
	p.fecha_registro ASC
